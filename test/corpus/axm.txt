query GetUser($id: UUID) -> User? {
    SELECT id, email
    FROM users
    WHERE id = $id;
}

query DeleteUser($id: UUID) {
    DELETE FROM users
    WHERE id = $id;
}

query ListUsers($limit: Int) -> User[] {
    SELECT *
    FROM users
    ORDER BY id
    LIMIT $limit;
}
